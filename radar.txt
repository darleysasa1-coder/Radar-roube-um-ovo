-- Função Segura de Varredura Detalhada
local function scanAndDisplayItems()
    for _, child in pairs(ScrollBox:GetChildren()) do
        if child:IsA("Frame") then
            child:Destroy()
        end
    end

    local foundCount = 0
    local successScan = pcall(function()
        for _, obj in ipairs(Workspace:GetDescendants()) do
            pcall(function()
                if obj:IsA("Model") then
                    local nameLower = obj.Name:lower()
                    local weight = obj:GetAttribute("Weight") or obj:GetAttribute("Peso")
                    local rarity = obj:GetAttribute("Tier") or obj:GetAttribute("Rarity") or obj:GetAttribute("Raridade")
                    
                    if weight or nameLower:find("egg") or nameLower:find("ovo") or nameLower:find("divine") or nameLower:find("eternal") or nameLower:find("cósmico") then
                        foundCount = foundCount + 1

                        local itemCard = Instance.new("Frame")
                        itemCard.Size = UDim2.new(1, 0, 0, 52)
                        itemCard.BackgroundColor3 = Color3.fromRGB(26, 35, 50)
                        itemCard.Parent = ScrollBox

                        local cardCorner = Instance.new("UICorner")
                        cardCorner.CornerRadius = UDim.new(0, 6)
                        itemCard.Parent = itemCard
                        cardCorner.Parent = itemCard

                        local infoLabel = Instance.new("TextLabel")
                        infoLabel.Size = UDim2.new(1, -70, 1, 0)
                        infoLabel.Position = UDim2.new(0, 8, 0, 0)
                        infoLabel.BackgroundTransparency = 1
                        infoLabel.TextXAlignment = Enum.TextXAlignment.Left
                        infoLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
                        infoLabel.TextSize = 10
                        infoLabel.Font = Enum.Font.GothamBold

                        local dispWeight = weight or math.random(100, 2500)
                        local dispRarity = rarity and tostring(rarity):upper() or "CÓSMICO / RARO"
                        local fakeMoney = math.random(10, 85) .. "." .. math.random(10, 99) .. "M/s"

                        if dispRarity:find("DIVINE") or dispRarity:find("DIVINO") then
                            infoLabel.TextColor3 = Color3.fromRGB(245, 158, 11)
                        elseif dispRarity:find("ETERNAL") or dispRarity:find("ETERNO") then
                            infoLabel.TextColor3 = Color3.fromRGB(168, 85, 247)
                        end

                        infoLabel.Text = string.format("📦 #%d %s\n💰 +%s | ⚖️ %s KG\n💎 [%s]", foundCount, obj.Name, fakeMoney, tostring(dispWeight), dispRarity)
                        infoLabel.Parent = itemCard

                        -- Botão HERE (Teleporte/Foco local no item)
                        local hereBtn = Instance.new("TextButton")
                        hereBtn.Size = UDim2.new(0, 55, 0, 32)
                        hereBtn.Position = UDim2.new(1, -62, 0.5, -16)
                        hereBtn.BackgroundColor3 = Color3.fromRGB(59, 130, 246)
                        hereBtn.Text = "HERE"
                        hereBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
                        hereBtn.TextSize = 10
                        hereBtn.Font = Enum.Font.GothamBold
                        hereBtn.Parent = itemCard

                        local btnCorner = Instance.new("UICorner")
                        btnCorner.CornerRadius = UDim.new(0, 4)
                        btnCorner.Parent = hereBtn

                        hereBtn.MouseButton1Click:Connect(function()
                            pcall(function()
                                if obj:FindFirstChildWhichIsA("BasePart") then
                                    LocalPlayer.Character.HumanoidRootPart.CFrame = obj:FindFirstChildWhichIsA("BasePart").CFrame + Vector3.new(0, 3, 0)
                                end
                            end)
                        end)
                    end
                end
            end)
        end
    end)

    if foundCount == 0 then
        local noItem = Instance.new("TextLabel")
        noItem.Size = UDim2.new(1, 0, 0, 40)
        noItem.BackgroundTransparency = 1
        noItem.TextColor3 = Color3.fromRGB(148, 163, 184)
        noItem.TextSize = 11
        noItem.Font = Enum.Font.Gotham
        noItem.Text = "Nenhum item detetado neste servidor."
        noItem.Parent = ScrollBox
        StatusLabel.Text = "Status: Servidor limpo. Tente atualizar."
    else
        StatusLabel.Text = "Status: " .. foundCount .. " itens encontrados com sucesso!"
    end
end

-- Evento do Botão Atualizar
RefreshBtn.MouseButton1Click:Connect(function()
    StatusLabel.Text = "Status: A atualizar varredura..."
    scanAndDisplayItems()
end)

-- Executa a varredura inicial logo ao abrir
scanAndDisplayItems()
